## RED / 控制断言

```ts
const selected=selectEvidenceWindows({methodStart:2,methodEnd:421,
 anchorLines:[201,301],maxLines:80});
assert.ok(selected.ranges.some(r=>r.startLine<=201&&201<=r.endLine));
assert.ok(selected.ranges.some(r=>r.startLine<=301&&301<=r.endLine));
assert.ok(selected.ranges.reduce((n,r)=>n+r.endLine-r.startLine+1,0)<=80);
assert.deepEqual(selected.omittedAnchors,[]);
```
